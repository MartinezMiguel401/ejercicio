NOTAS ANTES DE EMPEZAR
- Crea el proyecto en Android Studio como "Empty Views Activity" en Java,
  con el nombre Listview221025 (así sale el título en la barra de arriba).
- En este archivo el paquete es: com.example.listview221025
  Si tu proyecto se llama distinto, cambia SOLO la primera línea "package ..."
  de cada .java por el nombre de tu paquete.
- IMÁGENES de la lista: copia 3 imágenes de deportes a app/src/main/res/drawable/
  con los nombres deporte1, deporte2, deporte3 (.png o .jpg, en minúsculas,
  sin espacios ni guiones).
- Para crear la actividad "segunda": clic derecho en el paquete >
  New > Activity > Empty Views Activity (se agrega sola al Manifest).
- Para crear el fragmento "Saludito": clic derecho en el paquete >
  New > Fragment > Fragment (Blank). Borra todo lo que trae de ejemplo
  (ARG_PARAM1, newInstance, onCreate, etc.) y deja solo lo que viene abajo.
- Para crear el menú: clic derecho en res > New > Android Resource File >
  Resource type: Menu, nombre: menucito.
- Para crear la palomita: clic derecho en res/drawable > New > Drawable
  Resource File, nombre: palomita, y pega el código de abajo.
  (O New > Vector Asset > clip art "check circle" y nómbralo palomita).
- Archivos que NO se modifican en ningún paso: build.gradle.kts,
  settings.gradle.kts, strings.xml, colors.xml, AndroidManifest.xml
  (el Manifest ya lo deja bien Android Studio al crear "segunda").
- Ubicación de cada archivo:
    .java          -> app/src/main/java/com/example/listview221025/
    layouts .xml   -> app/src/main/res/layout/
    menucito.xml   -> app/src/main/res/menu/
    palomita.xml   -> app/src/main/res/drawable/
    themes.xml     -> app/src/main/res/values/

package com.example.listview221025;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ListView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

public class MainActivity extends AppCompatActivity implements AdapterView.OnItemClickListener {

    ListView vista;
    String nombre [] = {"uno", "dos", "tres"};
    String comentarios [] = {"deportes 1", "deportes 2", "deportes 3"};
    int imagenes [] = {R.drawable.deporte1, R.drawable.deporte2, R.drawable.deporte3};

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });

        vista = findViewById(R.id.lista);
        vista.setOnItemClickListener(this);
        adaptador adaptador = new adaptador(getApplicationContext(),
                comentarios, nombre, imagenes);
        vista.setAdapter(adaptador);
    }

    @Override
    public void onItemClick(AdapterView<?> parent, View v, int position, long id) {
        //todo position 0 es el primer elemento de la lista
        if(position == 0){
            Intent intentito = new Intent(this, segunda.class);
            startActivity(intentito);
        }
    }
}

package com.example.listview221025;

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.BaseAdapter;
import android.widget.ImageView;
import android.widget.TextView;

//todo ponle lo de extends BaseAdapter
public class adaptador extends BaseAdapter {
    Context contexto;
    String comentarios [];
    String nombre [];
    int imagenes [];
    LayoutInflater inflater;

    //todo Constructor
    public adaptador(Context contexto, String[] comentarios, String[] nombre,
                     int[] imagenes) {
        this.contexto = contexto;
        this.comentarios = comentarios;
        this.nombre = nombre;
        this.imagenes = imagenes;
        this.inflater = LayoutInflater.from(contexto);
    }

    @Override
    public int getCount(){
        return comentarios.length;
    }

    @Override
    public Object getItem(int position) {
        return null;
    }

    @Override
    public long getItemId(int position) {
        return 0;
    }

    @Override
    public View getView(int position, View convertView, ViewGroup parent) {
        convertView = inflater.inflate(R.layout.basurita, null);
        TextView texto1 = convertView.findViewById(R.id.texto1);
        TextView texto2 = convertView.findViewById(R.id.texto2);
        ImageView imagen = convertView.findViewById(R.id.fotito);
        texto1.setText(nombre[position]);
        texto2.setText(comentarios[position]);
        imagen.setImageResource(imagenes[position]);
        return convertView;
    }
}

package com.example.listview221025;

import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;

import androidx.activity.EdgeToEdge;
import androidx.annotation.NonNull;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;
import androidx.fragment.app.Fragment;
import androidx.fragment.app.FragmentManager;
import androidx.fragment.app.FragmentTransaction;

public class segunda extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_segunda);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu){
        getMenuInflater().inflate(R.menu.menucito, menu);
        return super.onCreateOptionsMenu(menu);
    }

    @Override
    public boolean onOptionsItemSelected(@NonNull MenuItem item){
        if(item.getItemId() == R.id.palomita || item.getItemId() == R.id.pasaste){
            //id de la palomita de arriba || id del kebab
            Saludito fragmentito = new Saludito();
            reemplazar(fragmentito);
        }
        return super.onOptionsItemSelected(item);
    }

    private void reemplazar(Fragment fragment){
        FragmentManager manejador = getSupportFragmentManager();
        FragmentTransaction transaccion = manejador.beginTransaction();
        transaccion.replace(R.id.base, fragment);
        transaccion.commit();
    }
}

package com.example.listview221025;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;
import android.widget.Toast;

import androidx.fragment.app.Fragment;

public class Saludito extends Fragment implements View.OnClickListener {

    View vista;
    Button boton;
    //Tienes que borrar todo  lo que aparece arriba hasta aqui

    @Override
    public View onCreateView(LayoutInflater inflater, ViewGroup container,
                             Bundle savedInstanceState) {
        // Inflate the layout for this fragment
        vista = inflater.inflate(R.layout.fragment_saludito, container, false);
        boton = vista.findViewById(R.id.saludito);
        boton.setOnClickListener(this);
        return vista;
    }

    @Override
    public void onClick(View v) {
        String cadenita = ((Button)v).getText().toString();

        if(cadenita.equals("saludito")){
            Toast.makeText(getContext(), "hola", Toast.LENGTH_SHORT).show();
        }
    }
}
