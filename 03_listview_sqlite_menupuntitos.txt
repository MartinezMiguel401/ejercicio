NOTAS ANTES DE EMPEZAR
- Crea el proyecto en Android Studio como "Empty Views Activity" en Java,
  con el nombre examen (así sale el título "examen" en la barra de arriba).
- En este archivo el paquete es: com.example.examen
  Si tu proyecto se llama distinto, cambia SOLO la primera línea "package ..."
  de cada .java por el nombre de tu paquete.
- IMÁGENES de la lista: copia 2 imágenes a app/src/main/res/drawable/
  con los nombres deporte1 y deporte2 (.png o .jpg, en minúsculas, sin
  espacios ni guiones).
- Para crear las actividades "crudcito" y "menupuntitos": clic derecho en el
  paquete > New > Activity > Empty Views Activity.
- Para crear adaptador, Base y mensajitos: clic derecho en el paquete >
  New > Java Class.
- Para crear el menú: clic derecho en res > New > Android Resource File >
  Resource type: Menu, nombre: menucito.
- Archivos que NO se modifican en ningún paso: build.gradle.kts,
  settings.gradle.kts, strings.xml, colors.xml.
- Ubicación de cada archivo:
    .java          -> app/src/main/java/com/example/examen/
    layouts .xml   -> app/src/main/res/layout/
    menucito.xml   -> app/src/main/res/menu/
    themes.xml     -> app/src/main/res/values/
    AndroidManifest.xml -> app/src/main/

package com.example.examen;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ListView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

public class MainActivity extends AppCompatActivity implements AdapterView.OnItemClickListener {

    ListView vista;
    String nombre [] = {"a base", "a menu"};
    String comentarios [] = {"vale 5", "otro de 5"};
    int imagenes [] = {R.drawable.deporte1, R.drawable.deporte2};

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });

        vista = findViewById(R.id.lista);
        vista.setOnItemClickListener(this);
        adaptador adaptador = new adaptador(getApplicationContext(),
                comentarios, nombre, imagenes);
        vista.setAdapter(adaptador);
    }

    @Override
    public void onItemClick(AdapterView<?> parent, View v, int position, long id) {
        //todo position 0 es el primer elemento, position 1 es el segundo
        if(position == 0){
            Intent intentito = new Intent(this, crudcito.class);
            startActivity(intentito);
        }else if(position == 1){
            Intent intentito = new Intent(this, menupuntitos.class);
            startActivity(intentito);
        }
    }
}

Cómo funciona:
 ALTAS:     clic en "altas" -> el botón cambia a "agregar", se inhabilita
            "consultas" y se habilitan los 3 cuadros. Tecleas los datos y das
            clic en "agregar" -> se guarda en tablita, el botón vuelve a
            "altas", "consultas" se habilita, se borran los cuadros y se
            inhabilitan.
 CONSULTAS: clic en "consultas" -> se habilita solo el cuadro de clave, se
            inhabilita "altas" y el botón cambia a "buscar". Tecleas la clave
            y das clic en "buscar" -> si existe, aparecen título y costo; si
            no, sale un mensaje de que no existe. Luego todo regresa a como
            estaba.

OJO: la clave es TEXTO, por eso en el select va entre comillas simples:
     "... where clave='" + codigo + "'"

public class Base extends SQLiteOpenHelper {

    public Base(@Nullable Context context, @Nullable String name, @Nullable SQLiteDatabase.CursorFactory factory, int version) {
        super(context, name, factory, version);
    }

    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL("create table tablita(clave text primary key, titulo text, costo int)");
    }

    @Override
    public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {

    }
}

package com.example.examen;

import android.content.ContentValues;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

public class crudcito extends AppCompatActivity implements View.OnClickListener {

    EditText etclave, ettitulo, etcosto;
    Button botonalt, botoncons;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_crudcito);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });

        etclave = findViewById(R.id.eclave);
        ettitulo = findViewById(R.id.etitulo);
        etcosto = findViewById(R.id.ecosto);

        botonalt = findViewById(R.id.altas);
        botonalt.setOnClickListener(this);

        botoncons = findViewById(R.id.consultas);
        botoncons.setOnClickListener(this);
    }

    @Override
    public void onClick(View v) {
        String cadenita = ((Button)v).getText().toString();

        if(cadenita.equals("altas")){
            ayuditaa();
        }else if(cadenita.equals("agregar")){
            Base admin = new Base(this, "disquitos", null, 1);
            SQLiteDatabase basesita = admin.getWritableDatabase();
            String llave = etclave.getText().toString();
            String nombre = ettitulo.getText().toString();
            int precio = Integer.parseInt(etcosto.getText().toString());
            ContentValues registro = new ContentValues();
            registro.put("clave", llave);
            registro.put("titulo", nombre);
            registro.put("costo", precio);
            basesita.insert("tablita", null, registro);
            basesita.close();
            ayuditaab();
        }else if(cadenita.equals("consultas")){
            ayuditac();
        }else if(cadenita.equals("buscar")){
            Base admin = new Base(this, "disquitos", null, 1);
            SQLiteDatabase basesita = admin.getReadableDatabase();
            String codigo = etclave.getText().toString();
            //todo la clave es texto, por eso va entre comillas simples
            Cursor fila = basesita.rawQuery("select titulo, costo from tablita where clave='" + codigo + "'", null);
            if(fila.moveToFirst()){
                ettitulo.setText(fila.getString(0));
                etcosto.setText(fila.getString(1));
            }else{
                Toast.makeText(this, "No existe el registro", Toast.LENGTH_SHORT).show();
            }
            fila.close();
            basesita.close();
            ayuditacab();
        }
    }

    //todo rutina de altas: al dar clic en "altas"
    private void ayuditaa(){
        etclave.setEnabled(true);
        ettitulo.setEnabled(true);
        etcosto.setEnabled(true);
        botoncons.setEnabled(false);
        botonalt.setText("agregar");
    }

    //todo rutina de altas: después de "agregar"
    private void ayuditaab(){
        etclave.setText("");
        ettitulo.setText("");
        etcosto.setText("");
        etclave.setEnabled(false);
        ettitulo.setEnabled(false);
        etcosto.setEnabled(false);
        botoncons.setEnabled(true);
        botonalt.setText("altas");
        Toast.makeText(this, "Agregado", Toast.LENGTH_SHORT).show();
    }

    //todo rutina de consultas: al dar clic en "consultas"
    private void ayuditac(){
        etclave.setEnabled(true);
        ettitulo.setEnabled(false);
        etcosto.setEnabled(false);
        botonalt.setEnabled(false);
        botoncons.setText("buscar");
    }

    //todo rutina de consultas: después de "buscar"
    private void ayuditacab(){
        etclave.setEnabled(false);
        ettitulo.setEnabled(false);
        etcosto.setEnabled(false);
        botonalt.setEnabled(true);
        botoncons.setText("consultas");
    }
}

        String cadenita = ((Button)v).getText().toString();

        if(cadenita.equals("regreso")){
            Intent intentito = new Intent(this, MainActivity.class);
            startActivity(intentito);
        }
    }
}
